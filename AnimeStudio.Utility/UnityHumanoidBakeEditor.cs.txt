using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Reflection;
using UnityEditor;
using UnityEngine;
using UnityEngine.Animations;
using UnityEngine.Playables;

// Embedded by AnimeStudio and compiled only in the disposable local bake project.
public static class AnimeStudioBake
{
    [Serializable] public class Axes { public Quaternion pre, post; public Vector3 sign, min, max; public float length; }
    [Serializable] public class Node { public string path, name, human; public int parent; public Vector3 position, scale; public Quaternion rotation; public Axes axes; }
    [Serializable] public class ClipInput { public string file, name, overlay; public float sampleRate, stopTime; }
    [Serializable] public class Job { public Node[] nodes; public ClipInput[] clips; public float humanScale, armTwist, foreArmTwist, upperLegTwist, legTwist, armStretch, legStretch, feetSpacing; public bool hasTranslationDoF; }
    [Serializable] public class Pose { public float time; public Vector3[] positions, scales; public Quaternion[] rotations; }
    [Serializable] public class ClipOutput { public int index; public bool human; public string name, composite; public float maxBodyPositionError, maxBodyAngleError; public string[] unresolvedPaths; public Pose[] poses; }
    [Serializable] public class Result { public string unityVersion; public float humanScale; public ClipOutput[] clips; }

    public static void Run()
    {
        try
        {
            var job = JsonUtility.FromJson<Job>(File.ReadAllText("Assets/Input/job.json"));
            var nodes = new Transform[job.nodes.Length];
            for (int i = 0; i < nodes.Length; i++)
            {
                var n = job.nodes[i];
                if (n.parent >= i || n.parent < -1 || (i > 0 && n.parent < 0)) throw new Exception("Invalid parent ordering.");
                nodes[i] = new GameObject(n.name).transform;
                if (n.parent >= 0) nodes[i].SetParent(nodes[n.parent], false);
            }
            void ResetPose()
            {
                for (int i = 0; i < nodes.Length; i++)
                {
                    nodes[i].localPosition = job.nodes[i].position;
                    nodes[i].localRotation = job.nodes[i].rotation;
                    nodes[i].localScale = job.nodes[i].scale;
                }
            }
            ResetPose();
            var human = job.nodes.Where(n => !string.IsNullOrEmpty(n.human)).Select(n => new HumanBone {
                boneName = n.name, humanName = HumanTrait.BoneName[(int)Enum.Parse(typeof(HumanBodyBones), n.human)],
                limit = new HumanLimit { useDefaultValues = false, min = n.axes.min * Mathf.Rad2Deg, max = n.axes.max * Mathf.Rad2Deg, center = Vector3.zero, axisLength = n.axes.length }
            }).ToArray();
            var description = new HumanDescription { human = human,
                skeleton = job.nodes.Select(n => new SkeletonBone { name = n.name, position = n.position, rotation = n.rotation, scale = n.scale }).ToArray(),
                upperArmTwist = job.armTwist, lowerArmTwist = job.foreArmTwist, upperLegTwist = job.upperLegTwist, lowerLegTwist = job.legTwist,
                armStretch = job.armStretch, legStretch = job.legStretch, feetSpacing = job.feetSpacing, hasTranslationDoF = job.hasTranslationDoF };
            var avatar = AvatarBuilder.BuildHumanAvatar(nodes[0].gameObject, description);
            if (!avatar.isValid || !avatar.isHuman) throw new Exception("Unity could not reconstruct the source humanoid Avatar.");
            foreach (var node in job.nodes.Where(n => !string.IsNullOrEmpty(n.human)))
            {
                int bone = (int)(HumanBodyBones)Enum.Parse(typeof(HumanBodyBones), node.human);
                object Get(string name) => typeof(Avatar).GetMethod(name, BindingFlags.Instance | BindingFlags.NonPublic).Invoke(avatar, new object[] { bone });
                if (Quaternion.Angle(node.axes.pre, (Quaternion)Get("GetPreRotation")) > .1f || Quaternion.Angle(node.axes.post, (Quaternion)Get("GetPostRotation")) > .1f)
                    throw new Exception("Rebuilt humanoid joint axes differ from the source: " + node.human);
            }
            var animator = nodes[0].gameObject.AddComponent<Animator>(); animator.avatar = avatar;
            animator.applyRootMotion = false; animator.cullingMode = AnimatorCullingMode.AlwaysAnimate;
            using var handler = new HumanPoseHandler(avatar, nodes[0]);
            var hips = nodes[Array.FindIndex(job.nodes, n => n.human == "Hips")];
            var pathIndices = job.nodes.Select((n, i) => new { n.path, i }).ToDictionary(n => n.path, n => n.i);
            var results = new List<ClipOutput>();
            long totalBoneSamples = 0;
            for (int clipIndex = 0; clipIndex < job.clips.Length; clipIndex++)
            {
                var source = job.clips[clipIndex];
                var clip = AssetDatabase.LoadAssetAtPath<AnimationClip>(source.file);
                if (clip == null) throw new Exception("Could not import " + source.file);
                if (Mathf.Abs(clip.length - source.stopTime) > .0001f) throw new Exception("Imported .anim duration differs from the authored source: " + source.name);
                string compositeFile = null;
                if (!string.IsNullOrEmpty(source.overlay))
                {
                    var overlay = AssetDatabase.LoadAssetAtPath<AnimationClip>(source.overlay);
                    if (overlay == null) throw new Exception("Could not import secondary clip.");
                    string combinedName = overlay.name + "__WithBody";
                    var combined = new AnimationClip { name = combinedName, frameRate = source.sampleRate };
                    void Copy(AnimationClip input, bool secondary)
                    {
                        foreach (var binding in AnimationUtility.GetCurveBindings(input))
                        {
                            if (secondary && binding.type == typeof(Animator)) continue;
                            AnimationUtility.SetEditorCurve(combined, binding, AnimationUtility.GetEditorCurve(input, binding));
                        }
                        foreach (var binding in AnimationUtility.GetObjectReferenceCurveBindings(input))
                            AnimationUtility.SetObjectReferenceCurve(combined, binding, AnimationUtility.GetObjectReferenceCurve(input, binding));
                    }
                    Copy(clip, false); Copy(overlay, true);
                    AnimationUtility.SetAnimationEvents(combined, AnimationUtility.GetAnimationEvents(clip).Concat(AnimationUtility.GetAnimationEvents(overlay)).OrderBy(e => e.time).ToArray());
                    AnimationUtility.SetAnimationClipSettings(combined, AnimationUtility.GetAnimationClipSettings(clip));
                    compositeFile = "Assets/Input/Combined-" + clipIndex + ".anim";
                    AssetDatabase.CreateAsset(combined, compositeFile);
                    combined.name = combinedName; EditorUtility.SetDirty(combined); AssetDatabase.SaveAssets();
                    clip = combined;
                }
                if (!clip.isHumanMotion) { results.Add(new ClipOutput { index = clipIndex, human = false, poses = new Pose[0], unresolvedPaths = new string[0] }); continue; }
                if (source.sampleRate <= 0 || source.sampleRate > 240 || source.stopTime <= 0 || source.stopTime > 1200) throw new Exception("Invalid/bounded-out clip sampling interval.");
                int count = (int)Math.Floor(source.stopTime * source.sampleRate + .0001f) + 1;
                if ((count - 1) / source.sampleRate < source.stopTime - .00001f) count++;
                totalBoneSamples += (long)count * nodes.Length;
                if (totalBoneSamples > 2000000) throw new Exception("Bake exceeds two million bone samples; select fewer clips.");
                AnimationCurve Curve(string name) => AnimationUtility.GetEditorCurve(clip, EditorCurveBinding.FloatCurve("", typeof(Animator), name));
                var rootT = "xyz".Select(c => Curve("RootT." + c)).ToArray();
                var rootQ = "xyzw".Select(c => Curve("RootQ." + c)).ToArray();
                if (rootT.Any(c => c == null) || rootQ.Any(c => c == null)) throw new Exception("Humanoid clip is missing its authored body-root curves: " + source.name);
                var transformCurves = AnimationUtility.GetCurveBindings(clip).Where(b => b.type == typeof(Transform))
                    .Select(b => new { b.path, b.propertyName, curve = AnimationUtility.GetEditorCurve(clip, b) }).ToArray();
                var unresolved = transformCurves.Where(b => !pathIndices.ContainsKey(b.path)).Select(b => b.path).Distinct().ToArray();
                var groups = transformCurves.Where(b => pathIndices.ContainsKey(b.path)).GroupBy(b => b.path).ToArray();
                var poses = new List<Pose>();
                float maxBodyPositionError = 0, maxBodyAngleError = 0;
                var settings = AnimationUtility.GetAnimationClipSettings(clip); settings.loopTime = false;
                AnimationUtility.SetAnimationClipSettings(clip, settings);
                var graph = PlayableGraph.Create("AnimeStudioBake"); graph.SetTimeUpdateMode(DirectorUpdateMode.Manual);
                var playable = AnimationClipPlayable.Create(graph, clip); playable.SetApplyFootIK(false); playable.SetApplyPlayableIK(false);
                AnimationPlayableOutput.Create(graph, "Pose", animator).SetSourcePlayable(playable);
                graph.Play();
                try
                {
                for (int sample = 0; sample < count; sample++)
                {
                    float time = Mathf.Min(sample / source.sampleRate, source.stopTime);
                    ResetPose();
                    animator.Rebind(); ResetPose();
                    playable.SetTime(time); graph.Evaluate(0);
                    var evaluated = new HumanPose(); handler.GetHumanPose(ref evaluated);
                    // Evaluate the complete clip (including translation DoF), then retain
                    // authored body-root placement instead of controller/scene extraction.
                    var targetPosition = new Vector3(rootT[0].Evaluate(time), rootT[1].Evaluate(time), rootT[2].Evaluate(time)) * animator.humanScale;
                    var targetRotation = Normalize(new Quaternion(rootQ[0].Evaluate(time), rootQ[1].Evaluate(time), rootQ[2].Evaluate(time), rootQ[3].Evaluate(time)));
                    var correction = targetRotation * Quaternion.Inverse(evaluated.bodyRotation);
                    hips.position = targetPosition + correction * (hips.position - evaluated.bodyPosition * animator.humanScale);
                    hips.rotation = correction * hips.rotation;
                    // Apply explicit non-humanoid Transform curves after the humanoid solve.
                    foreach (var group in groups)
                    {
                        var bone = nodes[pathIndices[group.Key]];
                        var p = bone.localPosition; var q = bone.localRotation; var s = bone.localScale; var e = bone.localEulerAngles;
                        bool rotated = false, euler = false;
                        foreach (var binding in group)
                        {
                            int dot = binding.propertyName.LastIndexOf('.');
                            string property = dot >= 0 ? binding.propertyName.Substring(0, dot) : binding.propertyName;
                            int axis = dot >= 0 ? "xyzw".IndexOf(binding.propertyName[dot + 1]) : -1;
                            if (axis < 0) throw new Exception("Unsupported Transform property " + binding.propertyName);
                            float value = binding.curve.Evaluate(time);
                            switch (property)
                            {
                                case "m_LocalPosition": p[axis] = value; break;
                                case "m_LocalRotation": q[axis] = value; rotated = true; break;
                                case "m_LocalScale": s[axis] = value; break;
                                case "localEulerAnglesRaw": case "localEulerAnglesBaked": case "localEulerAngles": e[axis] = value; euler = true; break;
                                default: throw new Exception("Unsupported Transform property " + binding.propertyName);
                            }
                        }
                        bone.localPosition = p; bone.localScale = s;
                        if (rotated) bone.localRotation = Normalize(q); else if (euler) bone.localEulerAngles = e;
                    }
                    var verified = new HumanPose(); handler.GetHumanPose(ref verified);
                    maxBodyPositionError = Mathf.Max(maxBodyPositionError, Vector3.Distance(verified.bodyPosition * animator.humanScale, targetPosition));
                    maxBodyAngleError = Mathf.Max(maxBodyAngleError, Quaternion.Angle(verified.bodyRotation, targetRotation));
                    poses.Add(new Pose { time = time, positions = nodes.Select(t => t.localPosition).ToArray(), rotations = nodes.Select(t => t.localRotation).ToArray(), scales = nodes.Select(t => t.localScale).ToArray() });
                }
                }
                finally { graph.Destroy(); }
                if (maxBodyPositionError > .001f || maxBodyAngleError > .1f) throw new Exception("Baked body-root verification failed: " + source.name);
                results.Add(new ClipOutput { index = clipIndex, human = true, name = clip.name, composite = compositeFile, poses = poses.ToArray(), unresolvedPaths = unresolved,
                    maxBodyPositionError = maxBodyPositionError, maxBodyAngleError = maxBodyAngleError });
                Debug.Log("Baked " + source.name + ": " + poses.Count + " samples");
            }
            File.WriteAllText("baked.json", JsonUtility.ToJson(new Result { unityVersion = Application.unityVersion, humanScale = animator.humanScale, clips = results.ToArray() }));
        }
        catch (Exception ex) { Debug.LogException(ex); EditorApplication.Exit(1); }
    }
    private static Quaternion Normalize(Quaternion q)
    {
        float length = Mathf.Sqrt(Quaternion.Dot(q, q));
        if (float.IsNaN(length) || float.IsInfinity(length) || length < 1e-6f) throw new Exception("Invalid source quaternion.");
        return new Quaternion(q.x / length, q.y / length, q.z / length, q.w / length);
    }
}
